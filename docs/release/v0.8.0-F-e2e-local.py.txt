def api_login(phone):
    """Вход через API (код 000000 в dev) → access-токен."""
    _rq.post(f"{API}/auth/request-code", json={"phone": phone}, timeout=10).raise_for_status()
    r = _rq.post(f"{API}/auth/verify", json={"phone": phone, "code": "000000"}, timeout=10)
    r.raise_for_status()
    return r.json()["accessToken"]

def create_delivery_as_buyer(phone=PHONE_BUYER):
    """Покупатель создаёт запрос доставки через API (UI-часть — у водителя).

    Точки: своя buyer_point (создаётся при необходимости) + линия продавца
    (seed). Возвращает id запроса.
    """
    import json as _json
    token = None
    if phone == PHONE_BUYER and TOKENS_FILE.exists():
        token = _json.loads(TOKENS_FILE.read_text())["a"]  # UI-токен Марины
    if not token:
        token = api_login(phone)
    h = {"Authorization": f"Bearer {token}"}
    mine = _rq.get(f"{API}/geo/points/mine", headers=h, timeout=10).json()
    dropoff = next((p for p in mine if p["kind"] == "buyer_point"), None)
    if not dropoff:
        # у Игоря нет точки — создаём (как из UI «Моя точка»)
        r = _rq.post(
            f"{API}/geo/points",
            json={"kind": "buyer_point", "latitude": 47.2172, "longitude": 39.7437,
                  "label": "Ларёк у вокзала", "address": "Ростов-Главный, ларёк"},
            headers=h, timeout=10,
        )
        if r.status_code < 300:
            dropoff = r.json()
            log("buyer_point Игоря создана")
        else:
            print("⚠ не удалось создать точку:", r.status_code, r.text[:120])
            return None
    if not dropoff:
        print("⚠ у покупателя нет buyer_point:", mine)
        return None
    sellers = _rq.get(
        f"{API}/geo/points/nearby",
        params={"lat": dropoff["latitude"], "lng": dropoff["longitude"],
                "radiusMeters": 5000, "kind": "seller_line"},
        headers=h, timeout=10,
    ).json()
    pickup = sellers[0] if isinstance(sellers, list) and sellers else None
    if not dropoff or not pickup:
        print("⚠ нет геоточек для доставки (mine=%s, sellers=%s)" % (len(mine), len(sellers or [])))
        return None
    body = {
        "pickupPointId": pickup["id"],
        "dropoffPointId": dropoff["id"],
        "weightKg": 80,
        "description": "E2E: помидоры и огурцы — линия 1 → ларёк «Заря»",
    }
    r = _rq.post(f"{API}/deliveries", json=body, headers=h, timeout=10)
    if r.status_code >= 300:
        print("⚠ создание доставки:", r.status_code, r.text[:150])
        return None
    log(f"доставка создана через API: {r.json().get('id')}")
    return r.json().get("id")


STEP = {"n": 0}



def read_qr_uuid(page):
    """UUID из открытого QR: Pressable «скопировать» с текстом-UUID."""
    import re as _re
    els = page.locator('[role="button"]')
    for i in range(els.count()):
        t = els.nth(i).inner_text().strip()
        if _re.match(r'^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$', t):
            return t
    return None


def shot(page, name, dark=False):

TOKENS_FILE = Path("/tmp/e2e-tokens-016.json")


def save_tokens(page):
    """Токены Марины из UI-сессии — для API-вызовов и session-restore
    (экономит request-code: троттлер 1/60 c на телефон)."""
    data = page.evaluate(
        "() => ({ a: localStorage.getItem('agromoll.accessToken'),"
        " r: localStorage.getItem('agromoll.refreshToken') })"
    )
    if data and data.get("a"):
        TOKENS_FILE.write_text(__import__("json").dumps(data))
        log("токены покупателя сохранены (localStorage)")


def login_restore(browser):
    """Session-restore по токенам из TOKENS_FILE — без request-code."""
    import json
    if not TOKENS_FILE.exists():
        return None, None
    data = json.loads(TOKENS_FILE.read_text())
    ctx = browser.new_context(
        viewport={"width": 1280, "height": 900}, locale="ru-RU"
    )
    page = ctx.new_page()
    page.set_default_timeout(15000)
    page.goto(BASE + "/auth", wait_until="domcontentloaded")
    page.evaluate(
        "([a, r]) => { localStorage.setItem('agromoll.accessToken', a);"
        " localStorage.setItem('agromoll.refreshToken', r); }",
        [data["a"], data["r"]],
    )
    page.goto(BASE + "/delivery", wait_until="domcontentloaded")
    time.sleep(2.5)
    return ctx, page

def flow_driver(browser):
    log("ФЛОУ 3 — водитель: радар, взять заказ, QR забора/передачи, доставка")
    ctx, page = login(browser, PHONE_DRIVER22, "Доставщик")
    page.wait_for_selector('text=Радар', timeout=15000)
    # Живой запрос: покупатель только что создал доставку через API
    # (UI-гимнастика выбора геоточек — вне фокуса этого флоу)
    create_delivery_as_buyer()
    time.sleep(1.5)
    page.reload()
    time.sleep(2.0)
    shot(page, "driver-01-radar")

    take = page.get_by_role("button", name="Беру")
    if take.count():
        take.first.click()
        time.sleep(2.0)
        shot(page, "driver-02-order-taken")
    else:
        log("⚠ открытых заказов в радаре нет (TTL?) — проверьте сид")
        ctx.close()
        return None

    # Мои доставки — назначенный заказ
    page.goto(BASE + "/deliveries")
    time.sleep(1.5)
    page.wait_for_selector('text=Мои доставки', timeout=15000)
    shot(page, "driver-03-my-deliveries-assigned")

    # QR забора (показ продавцу)
    qr_pickup = page.get_by_role("button", name="QR забора")
    if qr_pickup.count():
        qr_pickup.first.click()
        time.sleep(0.8)
        shot(page, "driver-04-qr-pickup")
        pickup_uuid = read_qr_uuid(page)
        log(f"QR-код забора: {pickup_uuid}")
        page.get_by_role("button", name="Скрыть QR забора").first.click()
        time.sleep(0.5)

    # Выехал за товаром
    page.get_by_role("button", name="Выехал за товаром").first.click()
    time.sleep(2.0)
    shot(page, "driver-05-in-transit")

    # QR передачи (показ покупателю) — читаем UUID для сверки покупателем
    dropoff_uuid = None
    qr_drop = page.get_by_role("button", name="QR передачи")
    if qr_drop.count():
        qr_drop.first.click()
        time.sleep(0.8)
        shot(page, "driver-06-qr-dropoff")
        dropoff_uuid = read_qr_uuid(page)
        log(f"QR-код передачи (активный рейс): {dropoff_uuid}")

    # Покупатель сверяет код передачи ПОКА рейс in_transit (web-QrScanner —
    # ручной ввод UUID, ADR-015)
    if dropoff_uuid:
        Path("/tmp/e2e-dropoff-code.txt").write_text(dropoff_uuid)
        log(f"код передачи: {dropoff_uuid}")
        verify_dropoff_code(browser, dropoff_uuid)

    # Товар передан → delivered
    page.get_by_role("button", name="Товар передан").first.click()
    time.sleep(2.0)
    shot(page, "driver-07-delivered")
    ctx.close()
    return None



def verify_dropoff_code(browser, code):
    """Покупатель сверяет QR-код передачи водителя (web-версия QrScanner)."""
    log(f"ФЛОУ 3b — покупатель сверяет QR-код передачи ({code[:8]}…)")
    ctx, page = login_restore(browser)
    if page is None:
        log("⚠ нет сохранённых токенов — пропуск сверки (запустите buyer-флоу)")
        return None
    verify = page.get_by_role("button", name="Сверить код водителя (QR)")
    if verify.count():
        verify.first.click()
        time.sleep(0.8)
        page.fill('[placeholder="UUID кода передачи"]', code)
        page.get_by_role("button", name="Проверить код").click()
        time.sleep(1.0)
        shot(page, "buyer-09-qr-verified")
    else:
        log("⚠ сверка недоступна (рейс не in_transit?)")
    ctx.close()


def seeded_dropoff_code():
    """Код передачи сид-доставки Анны (assigned, линия 6) из БД стенда."""
    import subprocess
    out = subprocess.run(
        ["/tmp/pg/usr/lib/postgresql/17/bin/psql", "-h", "127.0.0.1", "-U", "agromoll",
         "-d", "agromoll", "-t", "-A", "-c",
         "SELECT \"dropoffCode\" FROM delivery_requests WHERE status = 'assigned' LIMIT 1;"],
        capture_output=True, text=True, timeout=10,
    ).stdout.strip()
    return out or None

    shot(page, "driver-08-hours-blocked")

    appeal_btn = page.get_by_role("button", name="Оспорить блокировку")
    if appeal_btn.count() == 0:
        appeal_btn = page.get_by_text("Оспорить блокировку")
    if appeal_btn.count() == 0:
        log("⚠ блокировка не найдена (уже разблокирована?)")
        ctx.close()
        return None
    appeal_btn.first.click()
    time.sleep(1.0)
    page.wait_for_selector('textarea, input[placeholder*="лучил" i]', timeout=10000)
    sheet = page.locator('input[placeholder*="случилось" i]')
    if sheet.count() == 0:
        sheet = page.locator("textarea")
    sheet.first.fill("Смены задвоены в журнале: в эти субботы я работал на линии 9, есть записи")
    shot(page, "driver-09-appeal-form")
    page.get_by_role("button", name="Отправить в арбитраж").last.click()
    time.sleep(2.5)
    shot(page, "driver-10-appeal-submitted")
    ctx.close()
    return None


# ─── ФЛОУ 6: арбитр — спор + апелляция ─────────────────────────────────────


def flow_arbit(browser):
    log("ФЛОУ 6 — арбитр: очередь споров, решение, апелляция 289-ФЗ")
    ctx, page = login(browser, PHONE_ARBITRATOR, role_hint="Арбитраж рынка")
    page.wait_for_selector('text=Апелляции 289-ФЗ', timeout=15000)
    time.sleep(1.5)
    shot(page, "arbitrator-01-queue-disputes")

    # Спор under_review (из сида): карточка в очереди → кейс → решение
    case = page.get_by_text("недогруз", exact=False)
    if case.count():
        case.first.click()
        time.sleep(1.5)
        shot(page, "arbitrator-02-dispute-chat")
        favor = page.get_by_role("button", name="Прав покупатель")
        if favor.count():
            favor.first.click()
            time.sleep(0.8)
            # под-выбор последствий для брони (первый вариант — отмена брони)
            outcome = page.get_by_role("button", name="Отменить бронь", exact=False)
            if outcome.count():
                outcome.first.click()
                time.sleep(0.5)
            page.get_by_role("button", name="Вынести решение").last.click()
            time.sleep(2.5)
            shot(page, "arbitrator-03-dispute-resolved")
        else:
            log("⚠ кнопка «Прав покупатель» не найдена (спор уже решён?)")
    else:
        log("⚠ спор under_review не найден (решён ранее?)")

    # Вкладка «Апелляции 289-ФЗ»: сначала вернуться из кейса в очередь
    back = page.get_by_role("button", name="К очереди")
    if back.count():
        back.first.click()
        time.sleep(1.2)
    else:
        page.goto(BASE + "/arbitration")
        time.sleep(1.8)
    page.get_by_role("button", name="Апелляции 289-ФЗ").first.click()
    time.sleep(1.8)
    shot(page, "arbitrator-04-appeals-queue")
    # карточка апелляции из ФЛОУ 5 — пара с ФХ «Донская долина» (легаси)
    appeal_card = page.get_by_text("Донская долина", exact=False)
    if appeal_card.count():
        appeal_card.first.click()
        time.sleep(1.5)
        take = page.get_by_role("button", name="Взять в работу")
        if take.count():
            take.first.click()
            time.sleep(2.0)
            shot(page, "arbitrator-05-appeal-taken")
        unblock = page.get_by_role("button", name="Снять блокировку")
        if unblock.count():
            unblock.first.click()
            time.sleep(0.8)
            page.get_by_role("button", name="Вынести решение").last.click()
            time.sleep(2.5)
            shot(page, "arbitrator-06-appeal-unblocked")
        else:
            log("⚠ «Снять блокировку» не найдена")
    else:
        log("⚠ апелляция из ФЛОУ 5 не найдена в очереди")
    ctx.close()
    return None

def flow_dark(browser):
    """Тёмная тема: ключевые экраны (лента покупателя, витрина продавца)."""
    log("ФЛОУ 8 — тёмная тема: лента и витрина")
    ctx, page = login(browser, "+79000000017", "Покупатель", dark=True)
    # приложение уважает системную тему; если нет — жмём переключатель
    # (Playwright-клик по анимируемому хедеру флотает — JS-клик надёжен)
    toggle = page.locator('[aria-label="Переключить тему"]')
    if toggle.count():
        try:
            toggle.first.click(timeout=4000)
        except Exception:
            p_js = page.evaluate(
                "document.querySelector('[aria-label=\"Переключить тему\"]') && document.querySelector('[aria-label=\"Переключить тему\"]').click()"
            )
            del p_js
        time.sleep(1.0)
    page.wait_for_selector('[data-testid="listings-grid"]', timeout=20000)
    time.sleep(1.5)
    shot(page, "dark-buyer-feed", dark=True)
    ctx.close()

    ctx, page = login(browser, "+79000000006", "Продавец", dark=True)
    toggle = page.locator('[aria-label="Переключить тему"]')
    if toggle.count():
        try:
            toggle.first.click(timeout=4000)
        except Exception:
            page.evaluate(
                "document.querySelector('[aria-label=\"Переключить тему\"]') && document.querySelector('[aria-label=\"Переключить тему\"]').click()"
            )
        time.sleep(1.0)
    page.wait_for_selector('text=Витрина', timeout=15000)
    time.sleep(1.2)
    shot(page, "dark-seller-showcase", dark=True)
    ctx.close()
    return None


FLOWS = {
    "buyer": flow_buyer,
    "seller": flow_seller,
    "driver": flow_driver,
    "qr": flow_buyer_verify_qr,
    "debtor": flow_debtor,
    "appeal": flow_appeal,
    "arbit": flow_arbit,
    "video": flow_video,
    "dark": flow_dark,
}


def run(selected):
    with sync_playwright() as pw:
        browser = pw.chromium.launch()
        try:
            if selected == "all":
                # порядок важен: покупатель бронирует → продавец подтверждает;
                # водитель доставляет → покупатель сверяет; апелляция → арбитр
                for name in ["buyer", "seller", "driver", "qr", "debtor", "appeal", "arbit", "dark", "video"]:
                    log(f"=== {name.upper()} ===")
                    FLOWS[name](browser)
            else:
                FLOWS[selected](browser)
        finally:
            browser.close()


if __name__ == "__main__":
    selected = sys.argv[1] if len(sys.argv) > 1 else "buyer"
    run(selected)
    print("DONE")
